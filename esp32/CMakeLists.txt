cmake_minimum_required(VERSION 3.22)

set(IDF_TARGET esp32s3 CACHE STRING "TinyDraw firmware target")
set(TINYDRAW_FIRMWARE_VARIANT "product" CACHE STRING "TinyDraw firmware variant")
set_property(CACHE TINYDRAW_FIRMWARE_VARIANT PROPERTY STRINGS
             product demo raster-v1 gate qemu panel-probe tearing-probe tile-census
             touch-identity-probe)
if(NOT TINYDRAW_FIRMWARE_VARIANT MATCHES
   "^(product|demo|raster-v1|gate|qemu|panel-probe|tearing-probe|tile-census|touch-identity-probe)$")
  message(FATAL_ERROR
          "TINYDRAW_FIRMWARE_VARIANT must be product, demo, raster-v1, gate, qemu, panel-probe, tearing-probe, tile-census, or touch-identity-probe")
endif()

set(TINYDRAW_QEMU OFF)
set(TINYDRAW_RASTER_V1 OFF)
set(TINYDRAW_VECTOR_V2_GATE_HARNESS OFF)
set(TINYDRAW_VECTOR_V2_TILE_CENSUS OFF)
set(TINYDRAW_VECTOR_V2_TEARING_PROBE OFF)
set(TINYDRAW_VECTOR_V2_DEMO OFF)
set(TINYDRAW_PANEL_PROBE OFF)
set(TINYDRAW_TOUCH_IDENTITY_PROBE OFF)
if(TINYDRAW_FIRMWARE_VARIANT STREQUAL "qemu")
  set(TINYDRAW_QEMU ON)
elseif(TINYDRAW_FIRMWARE_VARIANT STREQUAL "raster-v1")
  set(TINYDRAW_RASTER_V1 ON)
elseif(TINYDRAW_FIRMWARE_VARIANT STREQUAL "panel-probe")
  set(TINYDRAW_PANEL_PROBE ON)
elseif(TINYDRAW_FIRMWARE_VARIANT STREQUAL "touch-identity-probe")
  set(TINYDRAW_TOUCH_IDENTITY_PROBE ON)
elseif(TINYDRAW_FIRMWARE_VARIANT STREQUAL "demo")
  set(TINYDRAW_VECTOR_V2_DEMO ON)
else()
  if(NOT TINYDRAW_FIRMWARE_VARIANT STREQUAL "product")
    set(TINYDRAW_VECTOR_V2_GATE_HARNESS ON)
  endif()
  if(TINYDRAW_FIRMWARE_VARIANT STREQUAL "tearing-probe")
    set(TINYDRAW_VECTOR_V2_TEARING_PROBE ON)
  elseif(TINYDRAW_FIRMWARE_VARIANT STREQUAL "tile-census")
    set(TINYDRAW_VECTOR_V2_TILE_CENSUS ON)
  endif()
endif()
option(TINYDRAW_QEMU_GRAPHICS "Enable the QEMU-only virtual display" OFF)
set(TINYDRAW_VECTOR_V2_TILE_SLOTS "604" CACHE STRING "Vector V2 raw tile cache slots (A/B builds only; product default 604)")
if(TINYDRAW_FIRMWARE_VARIANT MATCHES "^(product|demo)$")
  # Product capacity is fixed. This also migrates build trees that cached the
  # obsolete 448-slot CMake default while the header supplied 604.
  set(TINYDRAW_VECTOR_V2_TILE_SLOTS "604" CACHE STRING
      "Vector V2 raw tile cache slots (A/B builds only; product default 604)" FORCE)
endif()
if(TINYDRAW_QEMU_GRAPHICS AND NOT TINYDRAW_QEMU)
  message(FATAL_ERROR "TINYDRAW_QEMU_GRAPHICS requires the qemu firmware variant")
endif()
message(STATUS "TinyDraw firmware variant=${TINYDRAW_FIRMWARE_VARIANT}")
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
if(TINYDRAW_QEMU)
  set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.qemu.defaults")
else()
  set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
  if(TINYDRAW_VECTOR_V2_GATE_HARNESS)
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.gateharness.defaults")
  endif()
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(tinydraw_esp32)
