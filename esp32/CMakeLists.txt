cmake_minimum_required(VERSION 3.22)

set(IDF_TARGET esp32s3 CACHE STRING "TinyDraw firmware target")
option(TINYDRAW_QEMU "Build against QEMU's quad-PSRAM model" OFF)
option(TINYDRAW_QEMU_GRAPHICS "Enable the QEMU-only virtual display" OFF)
option(TINYDRAW_VECTOR_BENCHMARK "Run vector viewport benchmarks after startup" OFF)
option(TINYDRAW_PHASE2_PROTOTYPE "Run cached-pan/progressive-zoom prototype after startup" OFF)
option(TINYDRAW_RASTER_PAN_BENCHMARK "Measure the Raster V1 pan path after startup" OFF)
option(TINYDRAW_INTERACTIVE_PAN_BENCHMARK "Run the interactive vector-cache pan prototype" OFF)
option(TINYDRAW_VECTOR_V2_MEMORY_PROBE "Run the Vector V2 allocation receipt probe" OFF)
option(TINYDRAW_VECTOR_V2_OVERVIEW_WALK "Run the exclusive Vector V2 overview display walk" OFF)
option(TINYDRAW_VECTOR_V2_APP "Run the interactive Vector V2 application" OFF)
option(TINYDRAW_VECTOR_V2_GATE_HARNESS "Run Gate 1 before the interactive Vector V2 app" OFF)
option(TINYDRAW_VECTOR_V2_TILE_CENSUS "Measure Vector V2 tile payload classes" OFF)
option(TINYDRAW_VECTOR_V2_RASTER_CENSUS "Compile raster census counters into the firmware" OFF)
option(TINYDRAW_VECTOR_V2_TEARING_PROBE "Run a deterministic Vector V2 pan visual probe" OFF)
option(TINYDRAW_PANEL_PROBE "Run the CO5300 hardware characterization probe" OFF)
set(TINYDRAW_VECTOR_V2_PRESENTATION_EXPERIMENT "boundary-top-sweep" CACHE STRING
    "Vector V2 pan presentation experiment")
set_property(CACHE TINYDRAW_VECTOR_V2_PRESENTATION_EXPERIMENT PROPERTY STRINGS
             boundary-top-sweep beam-race-control)
if(NOT TINYDRAW_VECTOR_V2_PRESENTATION_EXPERIMENT MATCHES
   "^(boundary-top-sweep|beam-race-control)$")
  message(FATAL_ERROR
          "TINYDRAW_VECTOR_V2_PRESENTATION_EXPERIMENT must be boundary-top-sweep or beam-race-control")
endif()
set(TINYDRAW_VECTOR_V2_TE_EDGE "rising" CACHE STRING "Vector V2 selected TE edge")
set_property(CACHE TINYDRAW_VECTOR_V2_TE_EDGE PROPERTY STRINGS rising falling)
if(NOT TINYDRAW_VECTOR_V2_TE_EDGE MATCHES "^(rising|falling)$")
  message(FATAL_ERROR "TINYDRAW_VECTOR_V2_TE_EDGE must be rising or falling")
endif()
message(STATUS
        "Vector V2 presentation experiment=${TINYDRAW_VECTOR_V2_PRESENTATION_EXPERIMENT}, TE edge=${TINYDRAW_VECTOR_V2_TE_EDGE}")
set(TINYDRAW_VECTOR_V2_TILE_SLOTS "448" CACHE STRING "Vector V2 raw tile cache slots (A/B builds only; product default 448)")
if(TINYDRAW_VECTOR_V2_GATE_HARNESS AND NOT TINYDRAW_VECTOR_V2_APP)
  message(FATAL_ERROR "TINYDRAW_VECTOR_V2_GATE_HARNESS requires TINYDRAW_VECTOR_V2_APP")
endif()
if(TINYDRAW_VECTOR_V2_TILE_CENSUS AND NOT TINYDRAW_VECTOR_V2_GATE_HARNESS)
  message(FATAL_ERROR "TINYDRAW_VECTOR_V2_TILE_CENSUS requires TINYDRAW_VECTOR_V2_GATE_HARNESS")
endif()
if(TINYDRAW_VECTOR_V2_TEARING_PROBE AND NOT TINYDRAW_VECTOR_V2_GATE_HARNESS)
  message(FATAL_ERROR "TINYDRAW_VECTOR_V2_TEARING_PROBE requires TINYDRAW_VECTOR_V2_GATE_HARNESS")
endif()
set(TINYDRAW_BENCHMARK_MODE_COUNT 0)
foreach(TINYDRAW_BENCHMARK_MODE
    TINYDRAW_VECTOR_BENCHMARK
    TINYDRAW_PHASE2_PROTOTYPE
    TINYDRAW_RASTER_PAN_BENCHMARK
    TINYDRAW_INTERACTIVE_PAN_BENCHMARK
    TINYDRAW_VECTOR_V2_MEMORY_PROBE
    TINYDRAW_VECTOR_V2_OVERVIEW_WALK
    TINYDRAW_VECTOR_V2_APP)
  if(${TINYDRAW_BENCHMARK_MODE})
    math(EXPR TINYDRAW_BENCHMARK_MODE_COUNT "${TINYDRAW_BENCHMARK_MODE_COUNT} + 1")
  endif()
endforeach()
if(TINYDRAW_BENCHMARK_MODE_COUNT GREATER 1)
  message(FATAL_ERROR "Enable only one TinyDraw benchmark mode per build")
endif()
if(TINYDRAW_QEMU AND (TINYDRAW_VECTOR_V2_OVERVIEW_WALK OR TINYDRAW_VECTOR_V2_APP))
  message(FATAL_ERROR "Vector V2 hardware modes require physical hardware")
endif()
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
if(TINYDRAW_QEMU)
  set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.qemu.defaults")
else()
  set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
  if(TINYDRAW_VECTOR_V2_GATE_HARNESS)
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.gateharness.defaults")
  endif()
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(tinydraw_esp32)
